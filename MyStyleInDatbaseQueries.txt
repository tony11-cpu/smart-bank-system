ALTER PROCEDURE [dbo].[sp_Transfer]
(
    @UserInActionID   INT,
    @FromAccountID    INT,
    @ToAccountID      INT,
    @Amount           DECIMAL(18,2),
    @Description      NVARCHAR(250),
    @NewTransactionID INT OUTPUT
)
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRY
        BEGIN TRANSACTION

            DECLARE @FromBalanceAfter DECIMAL(18,2);
            DECLARE @ToBalanceAfter   DECIMAL(18,2);

            SELECT @FromBalanceAfter = Balance - @Amount
            FROM Accounts
            WHERE AccountID = @FromAccountID;

            SELECT @ToBalanceAfter = Balance + @Amount
            FROM Accounts
            WHERE AccountID = @ToAccountID;

            UPDATE Accounts
            SET Balance = @FromBalanceAfter
            WHERE AccountID = @FromAccountID;
            
            IF @@ROWCOUNT = 0
                THROW 52004, 'Source account update failed.', 1;
            
            UPDATE Accounts
            SET Balance = @ToBalanceAfter
            WHERE AccountID = @ToAccountID;
            
            IF @@ROWCOUNT = 0
                THROW 52005, 'Destination account update failed.', 1;

            INSERT INTO Transactions (AccountID, TransactionType, Amount,
                                      RelatedAccountID, Description, TransactionDate,
                                      ProcessedByUserID, IsScheduled)
            VALUES (@FromAccountID, 'Transfare', @Amount, @ToAccountID , @Description, GETDATE() , @UserInActionID, 0);

            SET @NewTransactionID = SCOPE_IDENTITY();

            IF @NewTransactionID IS NULL
                THROW 52003, 'Transfer transaction insertion failed.', 1;

            INSERT INTO AuditLog (UserID, Action, EntityType, EntityID, OldValue, NewValue, Timestamp)
            VALUES (@UserInActionID, 'TRANSFER', 'Accounts', @FromAccountID,
                    CAST(@FromAccountID AS NVARCHAR) + ' -> ' + CAST(@ToAccountID AS NVARCHAR),
                    CAST(@FromBalanceAfter AS NVARCHAR), GETDATE());

        COMMIT TRANSACTION
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;

        SET @NewTransactionID = -1;
        THROW;
    END CATCH
END;

-- LOCK
ALTER   PROCEDURE [dbo].[sp_LockUser]
(
    @AdminUserID INT,
    @UserID      INT,
    @IsUpdated   BIT OUTPUT
)
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRY
        BEGIN TRANSACTION;

        UPDATE Users
        SET IsLocked = 1
        WHERE UserID = @UserID;

        IF @@ROWCOUNT = 0
            THROW 51303, 'Lock User Failed.', 1;

        INSERT INTO AuditLog (UserID, Action, EntityType, EntityID, OldValue, NewValue, Timestamp)
        VALUES (@AdminUserID, 'USER_LOCKED', 'Users', @UserID, '0', '1', GETDATE());

        SET @IsUpdated = 1;
        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;

        SET @IsUpdated = 0;
        THROW;
    END CATCH
END;

    @AccountType      NVARCHAR(20)    OUTPUT,

ALTER   PROCEDURE [dbo].[sp_DeactivateCustomer]
    @CustomerID INT,
    @AdminUserID INT
AS
BEGIN
    SET NOCOUNT ON;

    BEGIN TRY
        BEGIN TRANSACTION

        UPDATE Customers
        SET IsActive = 0
        WHERE CustomerID = @CustomerID AND IsActive = 1;

        IF @@ROWCOUNT = 0
            THROW 51010, 'Customer not found or already inactive.', 1;

        INSERT INTO AuditLog (UserID, Action, EntityType, EntityID, OldValue, NewValue, Timestamp)
        VALUES (@AdminUserID, 'CUSTOMER_DEACTIVATED', 'Customers', @CustomerID, '1', '0', GETDATE());

        COMMIT TRANSACTION
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;

        THROW;
    END CATCH
END;

ALTER PROCEDURE [dbo].[sp_Withdraw]
(
    @UserInActionID   INT,
    @AccountID        INT,
    @Amount           DECIMAL(18,2),
    @Description      NVARCHAR(250),
    @NewTransactionID INT OUTPUT
)
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRY
        BEGIN TRANSACTION

            DECLARE @BalanceAfter DECIMAL(18,2);
            DECLARE @BalanceBefore DECIMAL(18,2);

            SELECT @BalanceBefore = Balance,
                   @BalanceAfter = Balance - @Amount
            FROM Accounts
            WHERE AccountID = @AccountID;

            UPDATE Accounts
            SET Balance = @BalanceAfter
            WHERE AccountID = @AccountID;

            IF @@ROWCOUNT = 0
               THROW 52004, 'account update failed.', 1;

            INSERT INTO Transactions (AccountID, TransactionType, Amount,
                                      RelatedAccountID, Description, TransactionDate,
                                      ProcessedByUserID, IsScheduled)
            VALUES (@AccountID, 'Withdrawal', @Amount,
                    NULL, @Description, GETDATE(),
                    @UserInActionID, 0);

            SET @NewTransactionID = SCOPE_IDENTITY();

            IF @NewTransactionID IS NULL
                THROW 52002, 'Withdrawal transaction insertion failed.', 1;

            INSERT INTO AuditLog (UserID, Action, EntityType, EntityID, OldValue ,NewValue, Timestamp)
            VALUES (@UserInActionID, 'WITHDRAWAL', 'Accounts', @AccountID, CAST(@BalanceBefore AS NVARCHAR) ,
            CAST(@BalanceAfter AS NVARCHAR), GETDATE());

        COMMIT TRANSACTION
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;

        SET @NewTransactionID = -1;
        THROW;
    END CATCH
END;

ALTER   Function [dbo].[fn_GetAllUserLoginAttempt] (@userID INT)
Returns Table 
AS 
Return
Select Top 4 l.AttemptDate as [Attempt Date],
       case l.WasSuccessful 
	      When 1 then 'Successful'
	      Else 'Failed'
	   End as [Login State]
From LoginAttempts l 
Where UserID = @userID
Order by l.AttemptDate desc
